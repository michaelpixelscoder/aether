def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def write(path,value):
    path.parent.mkdir(parents=True,exist_ok=True);path.write_text(json.dumps(value,indent=2)+'\n',encoding='utf-8')
def record(path):return {'file':path.relative_to(PACKAGE).as_posix(),'sha256':sha(path),'bytes':path.stat().st_size}
def cp(a,b):
    a=Path(a);b=Path(b);assert b.resolve().is_relative_to(PACKAGE.resolve())
    b.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(a,b);assert sha(a)==sha(b);return record(b)
def module(name,path):
    spec=importlib.util.spec_from_file_location(name,path)
    obj=importlib.util.module_from_spec(spec);spec.loader.exec_module(obj);return obj
